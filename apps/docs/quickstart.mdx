## 1. Write a plain service

```ts src/features/greeting/greeting.service.ts
export function createGreetingService(prefix: string) {
  return { greet: (name: string) => ({ message: `${prefix}, ${name}` }) };
}

export type GreetingService = ReturnType<typeof createGreetingService>;
```

## 2. Expose an HTTP boundary

```ts src/features/greeting/greeting.routes.ts
import { createApp, t } from "@kaonashi-dev/techne";
import type { GreetingService } from "./greeting.service";

export function greetingRoutes(service: GreetingService) {
  return createApp({ prefix: "/greetings" }).get(
    "/:name",
    {
      params: t.Object({ name: t.String({ minLength: 1 }) }),
    },
    ({ params }) => service.greet(params.name),
  );
}
```

Elysia 2 takes route options before the handler. The native schema infers the
handler's parameters. Return the fluent chain directly to retain route types.

## 3. Compose the application

```ts src/app.ts
import { createApp } from "@kaonashi-dev/techne";
import { createGreetingService } from "./features/greeting/greeting.service";
import { greetingRoutes } from "./features/greeting/greeting.routes";

export function buildApp(prefix = "Hello") {
  return createApp()
    .get("/healthz", () => ({ healthy: true }))
    .use(greetingRoutes(createGreetingService(prefix)));
}

export type App = ReturnType<typeof buildApp>;
```

## 4. Start explicitly

```ts src/main.ts
import { buildApp } from "./app";

const app = buildApp().listen({
  port: Number(Bun.env.PORT ?? 3000),
  hostname: Bun.env.HOST ?? "0.0.0.0",
});

let stopping: Promise<void> | undefined;
const stop = () => (stopping ??= Promise.resolve(app.stop()));
process.once("SIGTERM", stop);
process.once("SIGINT", stop);
```

```sh
bun run src/main.ts
curl http://localhost:3000/greetings/Ada
```

## 5. Test without a socket

```ts tests/app.test.ts
import { expect, test } from "bun:test";
import { buildApp } from "../src/app";

test("greets with the supplied dependency", async () => {
  const response = await buildApp("Hi").handle(new Request("http://localhost/greetings/Ada"));
  expect(response.status).toBe(200);
  expect(await response.json()).toEqual({ message: "Hi, Ada" });
});
```

Run `bun test`. See [native composition](/native-composition) for resources,
authorization, error mapping and the dependency rules used by larger features.
